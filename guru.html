⚙️ Pengaturan Sinkronisasi Google Sheets

Konfigurasi Webhook URL dan Kode Apps Script pendukung doGet & doPost.

📊 Buka Google Spreadsheet Langsung ➔

📋 Kode Apps Script Lengkap (doPost & doGet):

Kode ini memastikan hasil Pre-Test masuk ke Sheet Tab "Pre-Test" dan hasil Post-Test masuk ke Sheet Tab "Post-Test", sekaligus mengizinkan dashboard ini mengambil data secara otomatis melalui doGet. 

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var sheetName = (data.testType && data.testType.indexOf("Pre") !== -1) ? "Pre-Test" : "Post-Test";
    var sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }

    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Waktu Submit", "Tipe Ujian", "Nama Siswa", "Kelas", 
        "Skor (0-100)", "Jumlah Benar", "Total Soal", 
        "Soal 1 (Buku/Kantin)", "Soal 2 (Undo/Stack)", "Soal 3 (List Musik)", 
        "Soal 4 (Kardus Stack)", "Soal 5 (Klinik Queue)", "Soal 6 (Kontak List)"
      ]);
      var headerColor = (sheetName === "Pre-Test") ? "#0f3ba8" : "#ea580c";
      sheet.getRange(1, 1, 1, 13).setBackground(headerColor).setFontColor("#ffffff").setFontWeight("bold");
    }

    sheet.appendRow([
      data.timestamp || new Date().toLocaleString(),
      data.testType || "-",
      data.studentName || "-",
      data.studentClass || "-",
      data.score !== undefined ? data.score : "-",
      data.correctCount !== undefined ? data.correctCount : "-",
      data.totalQuestions || 6,
      data.q1 || "-", data.q2 || "-", data.q3 || "-",
      data.q4 || "-", data.q5 || "-", data.q6 || "-"
    ]);

    return ContentService.createTextOutput(JSON.stringify({ "status": "success", "sheet": sheetName }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var preSheet = ss.getSheetByName("Pre-Test");
    var postSheet = ss.getSheetByName("Post-Test");

    function readSheet(s, type) {
      if (!s || s.getLastRow() <= 1) return [];
      var vals = s.getDataRange().getValues();
      var arr = [];
      for (var i = 1; i < vals.length; i++) {
        var r = vals[i];
        arr.push({
          timestamp: r[0],
          testType: r[1] || type,
          studentName: r[2],
          studentClass: r[3],
          score: Number(r[4]) || 0,
          correctCount: Number(r[5]) || 0,
          totalQuestions: Number(r[6]) || 6,
          q1: r[7], q2: r[8], q3: r[9], q4: r[10], q5: r[11], q6: r[12]
        });
      }
      return arr;
    }

    var preRows = readSheet(preSheet, "Pre-Test");
    var postRows = readSheet(postSheet, "Post-Test");
    var combined = preRows.concat(postRows);

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      total: combined.length,
      preTest: preRows,
      postTest: postRows,
      data: combined
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
